# Build the real external FreeRTOS kernel using its GCC POSIX simulation port.
# Keep simulator tests separate from the hosted library ABI and physical-board timing claims.
cmake_minimum_required(VERSION 3.20)
project(rtx_freertos_sim LANGUAGES C CXX)
set(FREERTOS_KERNEL_PATH "" CACHE PATH "FreeRTOS-Kernel V11.2.0 source checkout")
if(NOT EXISTS "${FREERTOS_KERNEL_PATH}/tasks.c")
 message(FATAL_ERROR "Set FREERTOS_KERNEL_PATH to a FreeRTOS-Kernel checkout")
endif()
add_library(freertos_config INTERFACE)
target_include_directories(freertos_config INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
set(FREERTOS_PORT GCC_POSIX CACHE STRING "FreeRTOS port")
set(FREERTOS_HEAP 3 CACHE STRING "FreeRTOS heap")
add_subdirectory(${FREERTOS_KERNEL_PATH} kernel)
include(../../cmake/RTXFreeRTOS.cmake)
rtx_add_freertos_library(rtx_freertos freertos_kernel)
add_executable(freertos_test main.cpp)
target_link_libraries(freertos_test PRIVATE rtx_freertos)
enable_testing()
add_test(NAME freertos_kernel_execution COMMAND freertos_test)
set_tests_properties(freertos_kernel_execution PROPERTIES TIMEOUT 15)
add_executable(freertos_faults faults.cpp)
target_link_libraries(freertos_faults PRIVATE rtx_freertos)
foreach(scenario lifecycle watchdog runtime)
 add_test(NAME freertos_${scenario} COMMAND freertos_faults ${scenario})
 set_tests_properties(freertos_${scenario} PROPERTIES TIMEOUT 15)
endforeach()
# GNU linker wrapping injects a real task-creation failure without production hooks.
add_executable(freertos_allocation_fault allocation_fault.cpp)
target_link_libraries(freertos_allocation_fault PRIVATE rtx_freertos)
target_link_options(freertos_allocation_fault PRIVATE -Wl,--wrap=xTaskCreate)
add_test(NAME freertos_task_creation_failure COMMAND freertos_allocation_fault)
set_tests_properties(freertos_task_creation_failure PROPERTIES TIMEOUT 15)
