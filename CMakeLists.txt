# Export only enabled components so installed consumers inherit the correct dependencies.

project(rtx VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

option(RTX_POSIX "Build the Linux/POSIX execution and shared memory backend" ON)
option(RTX_GRPC "Build gRPC control service (requires RTX_POSIX)" ${RTX_POSIX})
option(RTX_TSAN "Enable ThreadSanitizer (separate from RTX_SANITIZE)" OFF)
if(RTX_TSAN AND RTX_SANITIZE)
 message(FATAL_ERROR "RTX_TSAN and RTX_SANITIZE require separate builds")
endif()
option(RTX_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(RTX_EXAMPLES "Build example applications and benchmark" ON)
if(RTX_GRPC AND NOT RTX_POSIX)
 message(FATAL_ERROR "RTX_GRPC currently requires RTX_POSIX")
endif()
find_package(Threads REQUIRED)
function(rtx_library target export_name)
 set_target_properties(${target} PROPERTIES EXPORT_NAME ${export_name} VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
 target_compile_features(${target} PUBLIC cxx_std_20)
 target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 if(RTX_TSAN)
  target_compile_options(${target} PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(${target} PUBLIC -fsanitize=thread)
 endif()
 if(RTX_SANITIZE)
  target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PUBLIC -fsanitize=address,undefined)
 endif()
 install(TARGETS ${target} EXPORT rtxTargets)
endfunction()

if(RTX_GRPC)

 set(RTX_PROTOC_EXECUTABLE "" CACHE FILEPATH "Host protoc executable for cross compilation")
 set(RTX_GRPC_PLUGIN_EXECUTABLE "" CACHE FILEPATH "Host grpc_cpp_plugin executable for cross compilation")
 if(RTX_PROTOC_EXECUTABLE)
  set(rtx_protoc "${RTX_PROTOC_EXECUTABLE}")
 else()
  set(rtx_protoc protobuf::protoc)
 endif()
 if(RTX_GRPC_PLUGIN_EXECUTABLE)
  set(rtx_grpc_plugin "${RTX_GRPC_PLUGIN_EXECUTABLE}")
 else()
  set(rtx_grpc_plugin "$<TARGET_FILE:gRPC::grpc_cpp_plugin>")
 endif()
 set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/proto/control.proto)
 add_custom_command(OUTPUT control.pb.cc control.pb.h control.grpc.pb.cc control.grpc.pb.h
 COMMAND ${rtx_protoc} ARGS --cpp_out=${CMAKE_CURRENT_BINARY_DIR} --grpc_out=${CMAKE_CURRENT_BINARY_DIR} -I${CMAKE_CURRENT_SOURCE_DIR}/proto --plugin=protoc-gen-grpc=${rtx_grpc_plugin} ${PROTO} DEPENDS ${PROTO})
 add_library(rtx_grpc src/control.cpp ${CMAKE_CURRENT_BINARY_DIR}/control.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/control.grpc.pb.cc)
 rtx_library(rtx_grpc grpc)
 add_library(rtx::grpc ALIAS rtx_grpc)
 target_include_directories(rtx_grpc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>)
 target_link_libraries(rtx_grpc PUBLIC rtx_posix gRPC::grpc++ protobuf::libprotobuf)
 install(FILES include/rtx/control.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/rtx)
 install(FILES ${CMAKE_CURRENT_BINARY_DIR}/control.pb.h ${CMAKE_CURRENT_BINARY_DIR}/control.grpc.pb.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
 install(FILES ${PROTO} DESTINATION ${CMAKE_INSTALL_DATADIR}/rtx/proto)
endif()
function(rtx_test name source library)
 add_executable(${name} ${source})
 target_link_libraries(${name} PRIVATE ${library})
 add_test(NAME ${name} COMMAND ${name})
 set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endfunction()
